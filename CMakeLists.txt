cmake_minimum_required(VERSION 3.24)
project(Blue VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
include(CTest)
option(BLUE_COVERAGE "Instrument first-party code with GCC coverage" OFF)
FetchContent_Declare(lua_src URL https://codeload.github.com/lua/lua/tar.gz/refs/tags/v5.4.8 URL_HASH SHA256=d85b70a65f43c5d2254944d58d625e822c8e2e10d9c6a3bd9b5b657e46376a19 SOURCE_SUBDIR unused)
FetchContent_Declare(json_src URL https://codeload.github.com/DaveGamble/cJSON/tar.gz/refs/tags/v1.7.19 URL_HASH SHA256=7fa616e3046edfa7a28a32d5f9eacfd23f92900fe1f8ccd988c1662f30454562 SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(lua_src json_src)
file(GLOB LUA_FILES "${lua_src_SOURCE_DIR}/*.c")
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
add_library(pack_deps STATIC ${LUA_FILES} "${json_src_SOURCE_DIR}/cJSON.c")
target_include_directories(pack_deps PUBLIC "${lua_src_SOURCE_DIR}" "${json_src_SOURCE_DIR}")
add_library(engine src/engine/engine.c src/engine/nodes.c src/engine/queue.c)
target_include_directories(engine PUBLIC include)
set(BLUE_EXTENSION_SOURCE "" CACHE FILEPATH "C source defining blue_register_extensions()")
if(BLUE_EXTENSION_SOURCE)
  target_sources(engine PRIVATE "${BLUE_EXTENSION_SOURCE}")
  target_compile_definitions(engine PRIVATE BLUE_WITH_EXTENSIONS)
endif()
if(NOT MSVC)
  target_link_libraries(engine PUBLIC m)
endif()
add_library(control src/control/pack.c src/control/wav.c)
target_link_libraries(control PUBLIC engine PRIVATE pack_deps)
if(BLUE_COVERAGE)
  foreach(t engine control)
    target_compile_options(${t} PRIVATE --coverage -O0)
  endforeach()
  add_link_options(--coverage)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_target(content ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/content" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/content")
if(EMSCRIPTEN)
  FetchContent_Declare(zip_src URL https://registry.npmjs.org/fflate/-/fflate-0.8.2.tgz URL_HASH SHA256=61fd5061e2fc8e5e3e3129f7f2fec7bd78a313e1bf4becbf1cc1cc9998d141dc SOURCE_SUBDIR unused)
  FetchContent_MakeAvailable(zip_src)
  add_executable(audio hosts/web/audio.c)
  target_link_libraries(audio PRIVATE engine)
  set_target_properties(audio PROPERTIES SUFFIX ".wasm")
  target_link_options(audio PRIVATE --no-entry -sSTANDALONE_WASM=1 -sALLOW_MEMORY_GROWTH=0 "-sEXPORTED_FUNCTIONS=['_audio_init','_audio_prepare','_audio_event','_audio_event_at','_audio_render','_audio_left','_audio_right','_audio_stage','_audio_commit','_audio_size']")
  add_executable(author hosts/web/author.c)
  target_link_libraries(author PRIVATE control)
  target_link_options(author PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_NAME=BlueAuthor -sENVIRONMENT=worker,node -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['FS','ccall','HEAPU8','UTF8ToString']" "-sEXPORTED_FUNCTIONS=['_author_load','_author_size','_author_label','_author_error']")
  add_custom_target(web ALL COMMAND ${CMAKE_COMMAND} -E copy "${zip_src_SOURCE_DIR}/esm/browser.js" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/zip-code.mjs" COMMAND ${CMAKE_COMMAND} -E copy "${zip_src_SOURCE_DIR}/LICENSE" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/zip-code.LICENSE" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/hosts/web/site" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
else()
  FetchContent_Declare(miniz URL https://codeload.github.com/richgel999/miniz/tar.gz/refs/tags/3.0.2 URL_HASH SHA256=c4b4c25a4eb81883448ff8924e6dba95c800094a198dc9ce66a292ac2ef8e018 SOURCE_SUBDIR unused)
  FetchContent_MakeAvailable(miniz)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h "#define MINIZ_EXPORT\n")
  add_library(miniz STATIC ${miniz_SOURCE_DIR}/miniz.c ${miniz_SOURCE_DIR}/miniz_zip.c ${miniz_SOURCE_DIR}/miniz_tinfl.c ${miniz_SOURCE_DIR}/miniz_tdef.c)
  target_include_directories(miniz PUBLIC ${miniz_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
  add_library(pack_archive hosts/archive.c)
  target_link_libraries(pack_archive PUBLIC control PRIVATE miniz pack_deps)
  add_executable(blue-pack hosts/pack_tool.c)
  target_link_libraries(blue-pack PRIVATE pack_archive)
  if(BLUE_COVERAGE)
    target_compile_options(pack_archive PRIVATE --coverage -O0)
    target_compile_options(blue-pack PRIVATE --coverage -O0)
  endif()
  add_executable(blue-render hosts/render.c)
  target_link_libraries(blue-render PRIVATE control)
  option(BLUE_DESKTOP "Build SDL desktop instrument" ON)
  if(BLUE_DESKTOP)
    if(BLUE_SDL_RUNTIME AND BLUE_SDL_INCLUDE)
      set(BLUE_SDL_TARGET sdl_runtime)
      add_library(sdl_runtime SHARED IMPORTED)
      set_target_properties(sdl_runtime PROPERTIES IMPORTED_LOCATION "${BLUE_SDL_RUNTIME}" INTERFACE_INCLUDE_DIRECTORIES "${BLUE_SDL_INCLUDE}")
    else()
      FetchContent_Declare(SDL URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-2.32.4 URL_HASH SHA256=e1e42bc0d686e145f1df85ea1ba980b2cb79ad475c62098b77337e0ae67d3080)
      set(SDL_SHARED OFF CACHE BOOL "" FORCE)
      set(SDL_STATIC ON CACHE BOOL "" FORCE)
      set(SDL_TEST OFF CACHE BOOL "" FORCE)
      FetchContent_MakeAvailable(SDL)
      set(BLUE_SDL_TARGET SDL2::SDL2-static)
    endif()
    FetchContent_Declare(pm URL https://codeload.github.com/PortMidi/portmidi/tar.gz/refs/tags/v2.0.4 URL_HASH SHA256=64893e823ae146cabd3ad7f9a9a9c5332746abe7847c557b99b2577afa8a607c)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(PM_USE_STATIC_RUNTIME OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(pm)
    add_executable(Blue hosts/desktop.c hosts/midi.c hosts/editor.c hosts/panel.c hosts/tuning.c)
    target_link_libraries(Blue PRIVATE pack_archive pack_deps control ${BLUE_SDL_TARGET} portmidi)
  endif()
  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(blue_lua_test_runner ${lua_src_SOURCE_DIR}/lua.c)
    target_link_libraries(blue_lua_test_runner PRIVATE pack_deps engine)
    file(GLOB_RECURSE FACTORY_LUA CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/content/*.lua")
    add_test(NAME lua_unit COMMAND blue_lua_test_runner ${CMAKE_SOURCE_DIR}/tests/lua_unit.lua ${CMAKE_SOURCE_DIR}/content/packs/blue.factory/scripts/performance.lua ${FACTORY_LUA})
    add_test(NAME coverage_gate_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/coverage_gate_unit.py $<TARGET_FILE:blue_lua_test_runner>)
    add_library(web_bindings STATIC hosts/web/audio.c hosts/web/author.c)
    target_link_libraries(web_bindings PRIVATE control)
    if(BLUE_COVERAGE)
      target_compile_options(web_bindings PRIVATE --coverage -O0)
    endif()
    add_test(NAME archive_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/archive_unit.py $<TARGET_FILE:blue-pack> ${CMAKE_SOURCE_DIR}/content/packs/blue.factory)
    add_executable(dsp_cases tests/dsp_cases.c)
    target_link_libraries(dsp_cases PRIVATE engine)
    add_test(NAME dsp_cases COMMAND dsp_cases)
    add_executable(adapter_unit tests/adapter_unit.c)
    target_link_libraries(adapter_unit PRIVATE web_bindings control)
    add_test(NAME web_c_unit COMMAND adapter_unit ${CMAKE_SOURCE_DIR}/content/packs/blue.factory)
    add_executable(render_unit tests/render_unit.c)
    target_link_libraries(render_unit PRIVATE control)
    if(BLUE_COVERAGE)
      target_compile_options(render_unit PRIVATE --coverage -O0)
    endif()
    file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/fixtures)
    add_test(NAME render_unit COMMAND render_unit ${CMAKE_SOURCE_DIR}/content/packs/blue.factory ${CMAKE_BINARY_DIR}/fixtures/export.wav ${CMAKE_BINARY_DIR}/fixtures/test.score)
    add_executable(control_unit tests/control_unit.c)
    target_link_libraries(control_unit PRIVATE control)
    add_test(NAME control_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/control_cases.py $<TARGET_FILE:control_unit>)
    if(BLUE_DESKTOP)
      add_executable(desktop_unit tests/desktop_unit.c hosts/editor.c hosts/panel.c hosts/tuning.c)
      target_link_libraries(desktop_unit PRIVATE pack_archive pack_deps control ${BLUE_SDL_TARGET})
      add_executable(midi_unit tests/midi_unit.c)
      target_include_directories(midi_unit PRIVATE include ${pm_SOURCE_DIR}/pm_common)
      if(BLUE_COVERAGE)
        target_compile_options(desktop_unit PRIVATE --coverage -O0)
        target_compile_options(midi_unit PRIVATE --coverage -O0)
      endif()
      file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
      add_test(NAME desktop_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/desktop_fixture.py $<TARGET_FILE:desktop_unit> ${CMAKE_SOURCE_DIR}/content)
      add_test(NAME midi_unit COMMAND midi_unit)
    endif()
    add_executable(blue_benchmark tests/benchmark.c)
    target_link_libraries(blue_benchmark PRIVATE control)
    add_executable(test_blue tests/unit.c)
    target_link_libraries(test_blue PRIVATE control)
    add_test(NAME unit COMMAND test_blue "${CMAKE_SOURCE_DIR}/content/packs/blue.factory")
  endif()
endif()

if(NOT EMSCRIPTEN)
  install(FILES ${miniz_SOURCE_DIR}/LICENSE DESTINATION licenses/miniz COMPONENT Blue)
  install(TARGETS blue-pack blue-render RUNTIME DESTINATION . COMPONENT Blue)
  if(BLUE_DESKTOP)
    install(TARGETS Blue RUNTIME DESTINATION . COMPONENT Blue)
    install(FILES ${pm_SOURCE_DIR}/license.txt DESTINATION licenses/portmidi COMPONENT Blue)
    if(EXISTS "${sdl_SOURCE_DIR}/LICENSE.txt")
      install(FILES ${sdl_SOURCE_DIR}/LICENSE.txt DESTINATION licenses/SDL COMPONENT Blue)
    endif()
  endif()
  install(DIRECTORY content/ DESTINATION content COMPONENT Blue)
  install(FILES ${json_src_SOURCE_DIR}/LICENSE DESTINATION licenses/cJSON COMPONENT Blue)
  install(FILES ${lua_src_SOURCE_DIR}/lua.h DESTINATION licenses/Lua COMPONENT Blue)
  install(FILES README.md DESTINATION . COMPONENT Blue)
endif()
