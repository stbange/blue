cmake_minimum_required(VERSION 3.24)
project(Blue VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
include(CTest)
option(BLUE_COVERAGE "Instrument first-party code with GCC coverage" OFF)
FetchContent_Declare(lua_src GIT_REPOSITORY https://github.com/lua/lua.git GIT_TAG v5.4.8 GIT_SHALLOW TRUE SOURCE_SUBDIR unused)
FetchContent_Declare(json_src GIT_REPOSITORY https://github.com/DaveGamble/cJSON.git GIT_TAG v1.7.19 GIT_SHALLOW TRUE SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(lua_src json_src)
file(GLOB LUA_FILES "${lua_src_SOURCE_DIR}/*.c")
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
add_library(pack_deps STATIC ${LUA_FILES} "${json_src_SOURCE_DIR}/cJSON.c")
target_include_directories(pack_deps PUBLIC "${lua_src_SOURCE_DIR}" "${json_src_SOURCE_DIR}")
add_library(engine src/engine/engine.c)
target_include_directories(engine PUBLIC include)
if(NOT MSVC)
  target_link_libraries(engine PUBLIC m)
endif()
add_library(control src/control/pack.c src/control/wav.c)
target_link_libraries(control PUBLIC engine PRIVATE pack_deps)
if(BLUE_COVERAGE)
  foreach(t engine control)
    target_compile_options(${t} PRIVATE --coverage -O0)
  endforeach()
  add_link_options(--coverage)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_custom_target(content ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/content" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/content")
if(EMSCRIPTEN)
  add_executable(audio hosts/web/audio.c)
  target_link_libraries(audio PRIVATE engine)
  set_target_properties(audio PROPERTIES SUFFIX ".wasm")
  target_link_options(audio PRIVATE --no-entry -sSTANDALONE_WASM=1 -sALLOW_MEMORY_GROWTH=0 "-sEXPORTED_FUNCTIONS=['_audio_init','_audio_event','_audio_render','_audio_left','_audio_right','_audio_stage','_audio_commit','_audio_size']")
  add_executable(author hosts/web/author.c)
  target_link_libraries(author PRIVATE control)
  target_link_options(author PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_NAME=BlueAuthor -sENVIRONMENT=worker,node -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['FS','ccall','HEAPU8','UTF8ToString']" "-sEXPORTED_FUNCTIONS=['_author_load','_author_size','_author_error']")
  add_custom_target(web ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/hosts/web/site" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
else()
  add_executable(blue-render hosts/render.c)
  target_link_libraries(blue-render PRIVATE control)
  option(BLUE_DESKTOP "Build SDL desktop instrument" ON)
  if(BLUE_DESKTOP)
    if(BLUE_SDL_RUNTIME AND BLUE_SDL_INCLUDE)
      add_library(sdl_runtime SHARED IMPORTED)
      set_target_properties(sdl_runtime PROPERTIES IMPORTED_LOCATION "${BLUE_SDL_RUNTIME}" INTERFACE_INCLUDE_DIRECTORIES "${BLUE_SDL_INCLUDE}")
    else()
      FetchContent_Declare(SDL GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.32.4 GIT_SHALLOW TRUE)
      set(SDL_SHARED OFF CACHE BOOL "" FORCE)
      set(SDL_STATIC ON CACHE BOOL "" FORCE)
      set(SDL_TEST OFF CACHE BOOL "" FORCE)
      FetchContent_MakeAvailable(SDL)
      add_library(sdl_runtime ALIAS SDL2::SDL2-static)
    endif()
    add_executable(Blue hosts/desktop.c)
    target_link_libraries(Blue PRIVATE control sdl_runtime)
  endif()
  if(BUILD_TESTING)
    add_executable(test_blue tests/unit.c)
    target_link_libraries(test_blue PRIVATE control)
    add_test(NAME unit COMMAND test_blue "${CMAKE_SOURCE_DIR}/content/packs/blue.factory")
  endif()
endif()
